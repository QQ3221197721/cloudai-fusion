139class OperationsAgent: 

140 """ 

141 LLM-powered operations agent for intelligent incident management. 

142 Uses LLM for root cause analysis and remediation suggestions, 

143 falls back to rule-based runbooks when LLM is unavailable. 

144 """ 

145 

146 def __init__(self, llm_client: LLMClient): 

147 self.llm = llm_client 

148 self.incident_history: List[IncidentAnalysis] = [] 

149 self.runbooks = self._load_runbooks() 

150 logger.info("operations_agent_initialized", llm_available=llm_client.available) 

151 

152 async def analyze_incident(self, incident: Incident, system_context: Optional[Dict] = None) -> IncidentAnalysis: 

153 """ 

154 Analyze an incident using LLM reasoning with rule-based fallback. 

155 """ 

156 start = time.time() 

157 context = system_context or self._default_system_context() 

158 

159 # Try LLM-powered analysis first 

160 analysis = await self._llm_analyze(incident, context) 

161 if analysis is None: 

162 # Fall back to rule-based analysis 

163 analysis = self._rule_based_analyze(incident) 

164 

165 self.incident_history.append(analysis) 

166 if len(self.incident_history) > 100: 

167 self.incident_history = self.incident_history[-100:] 

168 

169 latency = time.time() - start 

170 OPS_LATENCY.labels(action="analyze_incident").observe(latency) 

171 OPS_ACTIONS.labels(action="analyze_incident", method=analysis.analysis_method).inc() 

172 

173 return analysis

176 """Generate predictive scaling recommendations."""

206 OPS_ACTIONS.labels(action="scaling_recommendation", method=method).inc() 

207 

208 result["analysis_method"] = method 

209 return result

254 async def _llm_analyze(self, incident: Incident, context: dict) -> Optional[IncidentAnalysis]: 

255 """Use LLM for incident root cause analysis.""" 

256 messages = PromptTemplates.operations_incident(incident.to_dict(), context) 

257 result = await self.llm.chat_json(messages) 

258 

259 if result is None: 

260 return None 

261 

262 try: 

263 actions = [] 

264 for a in result.get("immediate_actions", []): 

265 actions.append( 

266 RemediationAction( 

267 action=a.get("action", ""), 

268 command=a.get("command", ""), 

269 risk=a.get("risk", "low"), 

270 requires_approval=a.get("risk", "low") in ("high", "critical"), 

271 ) 

272 ) 

273 

274 return IncidentAnalysis( 

275 incident_id=incident.incident_id, 

276 root_cause=result.get("root_cause_analysis", "LLM analysis inconclusive"), 

277 impact_assessment=result.get("impact_assessment", ""), 

278 immediate_actions=actions, 

279 remediation_script=result.get("remediation_script", ""), 

280 prevention_measures=result.get("prevention_measures", []), 

281 estimated_recovery_time=result.get("estimated_recovery_time", "unknown"), 

282 confidence=0.85, 

283 analysis_method="llm", 

284 ) 

285 except Exception as e: 

286 logger.warning("llm_analysis_parse_failed", error=str(e)) 

287 return None

293 def _rule_based_analyze(self, incident: Incident) -> IncidentAnalysis: 

294 """Fallback rule-based incident analysis using runbooks.""" 

295 runbook = self.runbooks.get(incident.category, self.runbooks["default"]) 

296 

297 actions = [RemediationAction(**a) for a in runbook["actions"]] 

298 

299 return IncidentAnalysis( 

300 incident_id=incident.incident_id, 

301 root_cause=runbook["root_cause_template"].format( 

302 resources=", ".join(incident.affected_resources[:3]), 

303 severity=incident.severity, 

304 ), 

305 impact_assessment=runbook["impact_template"].format( 

306 count=len(incident.affected_resources), 

307 ), 

308 immediate_actions=actions, 

309 remediation_script=runbook["script"], 

310 prevention_measures=runbook["prevention"], 

311 estimated_recovery_time=runbook["recovery_time"], 

312 confidence=0.65, 

313 analysis_method="rule_based", 

314 )

316 def _rule_based_scaling(self, metrics: Dict[str, Any]) -> Dict[str, Any]: 

317 """Rule-based scaling recommendations.""" 

318 recs = [] 

319 gpu_util = metrics.get("avg_gpu_utilization", 50) 

320 cpu_util = metrics.get("avg_cpu_utilization", 50) 

321 queue_depth = metrics.get("queue_depth", 0) 

322 

323 if gpu_util > 85: 

324 recs.append( 

325 { 

326 "resource": "gpu-node-pool", 

327 "action": "scale_up", 

328 "target": "+2 nodes", 

329 "reason": f"GPU utilization at {gpu_util}% exceeds 85% threshold", 

330 } 

331 ) 

332 elif gpu_util < 20 and queue_depth == 0: 

333 recs.append( 

334 { 

335 "resource": "gpu-node-pool", 

336 "action": "scale_down", 

337 "target": "-1 node", 

338 "reason": f"GPU utilization at {gpu_util}% with empty queue", 

339 } 

340 ) 

341 

342 if cpu_util > 80: 

343 recs.append( 

344 { 

345 "resource": "cpu-node-pool", 

346 "action": "scale_up", 

347 "target": "+3 nodes", 

348 "reason": f"CPU utilization at {cpu_util}%", 

349 } 

350 ) 

351 

352 if queue_depth > 10: 

353 recs.append( 

354 { 

355 "resource": "inference-replicas", 

356 "action": "scale_up", 

357 "target": f"+{min(queue_depth // 5, 10)} replicas", 

358 "reason": f"Queue depth {queue_depth} indicates demand spike", 

359 } 

360 ) 

361 

362 return { 

363 "recommendations": recs, 

364 "predicted_demand_change": "stable" if not recs else "increasing", 

365 "confidence": 0.6, 

366 }
