1""" 

2CloudAI Fusion - Anomaly Detection Benchmark. 

3 

4Reproducible, labeled evaluation of the multivariate MahalanobisDetector. It 

5generates a synthetic dataset of normal, correlated behavior plus injected 

6anomalies -- both extreme-magnitude outliers and JOINT/off-correlation outliers 

7whose per-feature marginals stay in range -- then reports standard 

8information-retrieval metrics (precision, recall, F1, ROC-AUC). 

9 

10This turns "anomaly detection works" into a measured, reproducible number rather 

11than a claim; tests/test_mahalanobis.py asserts the metrics in CI. 

12""" 

13 

14from __future__ import annotations 

15 

16from dataclasses import dataclass 

17 

18import numpy as np 

19from sklearn.metrics import roc_auc_score 

20 

21from anomaly.mahalanobis import MahalanobisDetector 

22 

23 

24@dataclass 

25class BenchmarkResult: 

26 """Information-retrieval metrics from a labeled anomaly-detection run.""" 

27 

28 precision: float 

29 recall: float 

30 f1: float 

31 roc_auc: float 

32 n_normal: int 

33 n_anomaly: int 

34 

35 

36def make_dataset(seed: int = 7, n_normal: int = 2000, n_anomaly: int = 200): 

37 """Generate a labeled multivariate dataset: correlated normals + injected anomalies. 

38 

39 Returns (x, y) where y == 1 marks anomalies. Half the anomalies are 

40 extreme-magnitude; half are JOINT anomalies with in-range marginals but a 

41 broken feature correlation (the case univariate detectors miss). 

42 """ 

43 rng = np.random.default_rng(seed) 

44 mean = np.array([60.0, 55.0, 65.0, 250.0]) 

45 std = np.array([12.0, 12.0, 8.0, 40.0]) 

46 corr = np.array([[1.0, 0.6, 0.5, 0.7], [0.6, 1.0, 0.4, 0.5], [0.5, 0.4, 1.0, 0.6], [0.7, 0.5, 0.6, 1.0]]) 

47 cov = corr * np.outer(std, std) 

48 normal = rng.multivariate_normal(mean, cov, size=n_normal) 

49 

50 half = n_anomaly // 2 

51 # Extreme-magnitude anomalies: far from the mean in every feature. 

52 extreme = rng.multivariate_normal(mean + 8.0 * std, cov * 0.25, size=half) 

53 # Joint anomalies: start from normal points, then break the util<->temp 

54 # correlation by pushing temperature strongly opposite to utilization. 

55 joint = normal[: n_anomaly - half].copy() 

56 joint[:, 2] = mean[2] - 4.0 * (joint[:, 0] - mean[0]) 

57 anomalies = np.vstack([extreme, joint]) 

58 

59 x = np.vstack([normal, anomalies]) 

60 y = np.concatenate([np.zeros(n_normal), np.ones(len(anomalies))]).astype(int) 

61 return x, y 

62 

63 

64def evaluate(seed: int = 7) -> BenchmarkResult: 

65 """Fit on a held-out normal baseline, score the rest, and compute IR metrics.""" 

66 x, y = make_dataset(seed=seed) 

67 normal_x = x[y == 0] 

68 n_train = int(len(normal_x) * 0.7) 

69 detector = MahalanobisDetector().fit(normal_x[:n_train]) 

70 

71 eval_normal = normal_x[n_train:] 

72 eval_x = np.vstack([eval_normal, x[y == 1]]) 

73 eval_y = np.concatenate([np.zeros(len(eval_normal)), np.ones(int(y.sum()))]).astype(int) 

74 

75 preds = detector.predict(eval_x).astype(int) 

76 scores = detector.score(eval_x) 

77 

78 tp = int(np.sum((preds == 1) & (eval_y == 1))) 

79 fp = int(np.sum((preds == 1) & (eval_y == 0))) 

80 fn = int(np.sum((preds == 0) & (eval_y == 1))) 

81 precision = tp / (tp + fp) if (tp + fp) > 0 else 0.0 

82 recall = tp / (tp + fn) if (tp + fn) > 0 else 0.0 

83 f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0.0 

84 roc_auc = float(roc_auc_score(eval_y, scores)) 

85 

86 return BenchmarkResult( 

87 precision=precision, 

88 recall=recall, 

89 f1=f1, 

90 roc_auc=roc_auc, 

91 n_normal=int(np.sum(eval_y == 0)), 

92 n_anomaly=int(np.sum(eval_y == 1)), 

93 )
