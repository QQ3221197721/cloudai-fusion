1""" 

2CloudAI Fusion - Distributed, Topology-Aware RL Scheduling Trainer. 

3 

4This module accelerates the RL scheduler (L10 Compute Well) by training multiple 

5Q-learning workers in parallel over sharded experience and aggregating their 

6tables via federated averaging. It is GPU-topology-aware: placement rewards and 

7action scoring account for NVLink domains so co-located, high-bandwidth GPUs are 

8preferred for communication-heavy workloads. 

9 

10Design (consistent with the platform's "optional heavy deps" pattern): 

11 - Core algorithm is pure NumPy + stdlib ``concurrent.futures`` so it runs in CI 

12 with no GPU, Ray, PyTorch, or Gymnasium. 

13 - An optional Ray backend (``try/except ImportError``) parallelizes across a 

14 cluster when available; otherwise a thread-pool backend is used and honestly 

15 reported. 

16 

17Usage: 

18 topo = GPUTopology(num_gpus=8, nvlink_domains=[[0, 1, 2, 3], [4, 5, 6, 7]]) 

19 trainer = ParallelQTrainer(topology=topo, preferred_domain=0) 

20 result = trainer.train(num_workers=4, episodes_per_worker=2000, seed=7) 

21 best_gpu = int(result.q_table[0].argmax()) 

22"""

48# GPU Topology

52@dataclass 

53class GPUTopology: 

54 """Describes the GPU interconnect: how many GPUs and which NVLink domains. 

55 

56 An NVLink domain is a set of GPU ids with high-bandwidth intra-domain links. 

57 Placement within the same domain is cheaper for communication-heavy jobs. 

58 """ 

59 

60 num_gpus: int 

61 nvlink_domains: list[list[int]] = field(default_factory=list) 

62 

63 def domain_of(self, gpu: int) -> int: 

64 """Return the NVLink domain index containing ``gpu`` (-1 if none).""" 

65 for idx, domain in enumerate(self.nvlink_domains): 

66 if gpu in domain: 

67 return idx 

68 return -1 

69 

70 def same_domain(self, gpu_a: int, gpu_b: int) -> bool: 

71 """Report whether two GPUs share an NVLink domain.""" 

72 da = self.domain_of(gpu_a) 

73 return da != -1 and da == self.domain_of(gpu_b) 

74 

75 

76def topology_affinity(topology: GPUTopology, gpu: int, preferred_domain: int) -> float: 

77 """Return a normalized affinity bonus for placing on ``gpu``. 

78 

79 1.0 when the GPU is in the preferred NVLink domain, else 0.0. Kept simple and 

80 deterministic so it composes predictably with learned Q-values. 

81 """ 

82 return 1.0 if topology.domain_of(gpu) == preferred_domain else 0.0

90@dataclass 

91class QLearnerConfig: 

92 """Hyperparameters for the tabular Q-learner.""" 

93 

94 alpha: float = 0.1 

95 gamma: float = 0.9 

96 epsilon: float = 0.1 

97 topology_weight: float = 0.5 

98 

99 

100class TopologyAwareQLearner: 

101 """Tabular Q-learner whose action scoring blends learned value and topology. 

102 

103 States and actions are both indexed by GPU id for the placement problem; the 

104 Q-table has shape (num_states, num_gpus). 

105 """ 

106 

107 def __init__(self, topology: GPUTopology, preferred_domain: int, config: Optional[QLearnerConfig] = None): 

108 self.topology = topology 

109 self.preferred_domain = preferred_domain 

110 self.config = config or QLearnerConfig() 

111 self.q_table = np.zeros((topology.num_gpus, topology.num_gpus), dtype=np.float64) 

112 

113 def score_actions(self, state: int) -> np.ndarray: 

114 """Return topology-blended scores for every action in ``state``.""" 

115 base = self.q_table[state] 

116 bonus = np.zeros(self.topology.num_gpus, dtype=np.float64) 

117 for gpu in range(self.topology.num_gpus): 

118 bonus[gpu] = topology_affinity(self.topology, gpu, self.preferred_domain) 

119 return base + self.config.topology_weight * bonus 

120 

121 def select_action(self, state: int, rng: np.random.Generator, explore: bool = True) -> int: 

122 """Epsilon-greedy action selection over the topology-blended scores.""" 

123 if explore and rng.random() < self.config.epsilon: 

124 return int(rng.integers(0, self.topology.num_gpus)) 

125 return int(self.score_actions(state).argmax()) 

126 

127 def update(self, state: int, action: int, reward: float, next_state: int) -> None: 

128 """Apply one Q-learning update.""" 

129 best_next = float(self.q_table[next_state].max()) 

130 target = reward + self.config.gamma * best_next 

131 td = target - self.q_table[state, action] 

132 self.q_table[state, action] += self.config.alpha * td

136# Federated aggregation 

137# =============================================================================

140def aggregate_q_tables(tables: list[np.ndarray], weights: Optional[list[float]] = None) -> np.ndarray: 

141 """Aggregate worker Q-tables via (optionally weighted) averaging. 

142 

143 Raises ValueError on an empty list or shape mismatch so a misconfigured run 

144 fails loudly rather than silently producing a wrong policy. 

145 """ 

146 if not tables: 

147 raise ValueError("aggregate_q_tables: no tables to aggregate") 

148 shape = tables[0].shape 

149 for t in tables: 

150 if t.shape != shape: 

151 raise ValueError(f"aggregate_q_tables: shape mismatch {t.shape} != {shape}") 

152 if weights is None: 

153 weights = [1.0] * len(tables) 

154 if len(weights) != len(tables): 

155 raise ValueError("aggregate_q_tables: weights length must match tables") 

156 total = float(sum(weights)) 

157 if total <= 0: 

158 raise ValueError("aggregate_q_tables: weights must sum to a positive value") 

159 acc = np.zeros(shape, dtype=np.float64) 

160 for t, w in zip(tables, weights): 

161 acc += t * w 

162 return acc / total

170@dataclass 

171class TrainResult: 

172 """Outcome of a parallel training run.""" 

173 

174 q_table: np.ndarray 

175 num_workers: int 

176 episodes: int 

177 backend: str # "ray" | "threads" 

178 

179 def best_action(self, state: int = 0) -> int: 

180 """Return the greedy action for a state under the aggregated table.""" 

181 return int(self.q_table[state].argmax())

184def _synthetic_reward(topology: GPUTopology, preferred_domain: int, action_gpu: int) -> float: 

185 """Deterministic placement reward: 1.0 inside the preferred NVLink domain. 

186 

187 This stands in for a real scheduling reward (utilization, locality, SLA) and 

188 makes training outcomes reproducible for tests and CI. 

189 """ 

190 return 1.0 if topology.domain_of(action_gpu) == preferred_domain else 0.0 

191 

192 

193def _train_worker(topology: GPUTopology, preferred_domain: int, episodes: int, seed: int) -> np.ndarray: 

194 """Train one worker for ``episodes`` and return its learned Q-table.""" 

195 rng = np.random.default_rng(seed) 

196 learner = TopologyAwareQLearner(topology, preferred_domain) 

197 state = 0 

198 for _ in range(episodes): 

199 action = learner.select_action(state, rng, explore=True) 

200 reward = _synthetic_reward(topology, preferred_domain, action) 

201 learner.update(state, action, reward, next_state=state) 

202 return learner.q_table

205class ParallelQTrainer: 

206 """Trains Q-learning workers in parallel and aggregates their tables. 

207 

208 The thread-pool backend is always available; the Ray backend is used only 

209 when Ray is importable, and the chosen backend is reported in the result. 

210 """ 

211 

212 def __init__(self, topology: GPUTopology, preferred_domain: int): 

213 if topology.num_gpus <= 0: 

214 raise ValueError("ParallelQTrainer: topology.num_gpus must be positive") 

215 self.topology = topology 

216 self.preferred_domain = preferred_domain 

217 

218 def train(self, num_workers: int = 4, episodes_per_worker: int = 1000, seed: int = 0) -> TrainResult: 

219 """Run ``num_workers`` workers and return the aggregated policy.""" 

220 num_workers = max(1, num_workers) 

221 seeds = [seed + i for i in range(num_workers)] 

222 

223 if _HAS_RAY: 

224 tables = self._train_ray(seeds, episodes_per_worker) 

225 backend = "ray" 

226 else: 

227 tables = self._train_threads(seeds, episodes_per_worker) 

228 backend = "threads" 

229 

230 merged = aggregate_q_tables(tables) 

231 logger.info( 

232 "distributed_rl_train_complete", 

233 backend=backend, 

234 workers=num_workers, 

235 episodes=num_workers * episodes_per_worker, 

236 preferred_domain=self.preferred_domain, 

237 ) 

238 return TrainResult( 

239 q_table=merged, 

240 num_workers=num_workers, 

241 episodes=num_workers * episodes_per_worker, 

242 backend=backend, 

243 ) 

244 

245 def _train_threads(self, seeds: list[int], episodes: int) -> list[np.ndarray]: 

246 """Thread-pool backend (always available).""" 

247 with ThreadPoolExecutor(max_workers=len(seeds)) as pool: 

248 futures = [pool.submit(_train_worker, self.topology, self.preferred_domain, episodes, s) for s in seeds] 

249 return [f.result() for f in futures] 

250 

251 def _train_ray(self, seeds: list[int], episodes: int) -> list[np.ndarray]: 

252 """Ray backend (used only when Ray is importable).""" 

253 if not ray.is_initialized(): 

254 ray.init(ignore_reinit_error=True, log_to_driver=False) 

255 remote_worker = ray.remote(_train_worker) 

256 refs = [remote_worker.remote(self.topology, self.preferred_domain, episodes, s) for s in seeds] 

257 return list(ray.get(refs))
