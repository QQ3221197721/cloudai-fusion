1""" 

2CloudAI Fusion - Model Provenance producer (Moat B / M1, Python side). 

3 

4This is the training-time producer for docs/verifiable-moat-spec.md M-B 

5(provenance-verifiable learning). It emits the CONTENT fields of a 

6cloudai-fusion ModelProvenance for a trained model; the Go control plane 

7(pkg/provenance) then Ed25519-SIGNS it and records a `model.provenance` receipt, 

8binding the weights to the exact signed, in-scope DatasetManifest they were 

9trained on. An auditor later runs `cafctl verify-model-provenance`. 

10 

11The field names match the Go struct's JSON tags EXACTLY so the control plane can 

12ingest and sign without translation: 

13 

14 base_model_hash, dataset_manifest_hash, train_config_hash, weights_hash, 

15 method, trainer, created_at 

16 

17(id / key_id / signature are filled in by the signer, Go-side.) This module is 

18deliberately dependency-free (stdlib only) so it runs wherever training runs, with 

19or without torch / stable-baselines3. 

20""" 

21 

22from __future__ import annotations 

23 

24import hashlib 

25import json 

26from datetime import datetime, timezone 

27from typing import Any, Dict 

28 

29# Hashes are hex-encoded SHA-256, matching the Go ledger's convention (raw hex, no 

30# prefix), so cross-language comparison is byte-exact. 

31_CHUNK = 1 << 16 

32 

33 

34def sha256_hex(data: bytes) -> str: 

35 """Hex SHA-256 of the given bytes.""" 

36 return hashlib.sha256(data).hexdigest() 

37 

38 

39def hash_file(path: str) -> str: 

40 """Hex SHA-256 of a file's contents, streamed (weights may be large).""" 

41 h = hashlib.sha256() 

42 with open(path, "rb") as f: 

43 for chunk in iter(lambda: f.read(_CHUNK), b""): 

44 h.update(chunk) 

45 return h.hexdigest() 

46 

47 

48def canonical_config_hash(config: Dict[str, Any]) -> str: 

49 """ 

50 Hex SHA-256 over the canonical JSON of a training config: sorted keys, compact 

51 separators, UTF-8, no HTML escaping - reproducible across runs and languages. 

52 """ 

53 b = json.dumps(config, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8") 

54 return sha256_hex(b) 

55 

56 

57# The exact key set the Go signer expects (its ModelProvenance content fields). 

58PROVENANCE_FIELDS = ( 

59 "base_model_hash", 

60 "dataset_manifest_hash", 

61 "train_config_hash", 

62 "weights_hash", 

63 "method", 

64 "trainer", 

65 "created_at", 

66)

69def build_model_provenance( 

70 *, 

71 base_model_hash: str, 

72 dataset_manifest_hash: str, 

73 weights_path: str, 

74 train_config: Dict[str, Any], 

75 method: str = "dpo", 

76 trainer: str = "advanced_trainer.py", 

77) -> Dict[str, Any]: 

78 """ 

79 Build the unsigned ModelProvenance content for a completed training run. 

80 

81 base_model_hash and dataset_manifest_hash are PASS-THROUGH inputs supplied by 

82 the control plane that started the run (the manifest hash is exactly what the 

83 Go DatasetManifest was committed as); weights_hash and train_config_hash are 

84 computed here from the produced artifacts. The result is ready to hand to the 

85 Go signer (provenance.SignProvenance + RecordProvenance). 

86 """ 

87 if not base_model_hash: 

88 raise ValueError("base_model_hash is required") 

89 if not dataset_manifest_hash: 

90 raise ValueError("dataset_manifest_hash is required (bind to the signed corpus)") 

91 return { 

92 "base_model_hash": base_model_hash, 

93 "dataset_manifest_hash": dataset_manifest_hash, 

94 "train_config_hash": canonical_config_hash(train_config), 

95 "weights_hash": hash_file(weights_path), 

96 "method": method, 

97 "trainer": trainer, 

98 "created_at": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"), 

99 } 

100 

101 

102def write_model_provenance(path: str, **kwargs: Any) -> Dict[str, Any]: 

103 """Build the provenance content and write it as JSON to path; returns the dict.""" 

104 prov = build_model_provenance(**kwargs) 

105 with open(path, "w", encoding="utf-8") as f: 

106 json.dump(prov, f, indent=2, sort_keys=True) 

107 return prov
