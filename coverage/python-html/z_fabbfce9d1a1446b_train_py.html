30class GPUSchedulingEnv: 

31 """ 

32 Custom Gymnasium environment for GPU workload scheduling. 

33 

34 State space: 

35 - Per-node: GPU utilization, memory usage, CPU usage, network bandwidth 

36 - Workload queue: type, GPU count, priority, framework 

37 

38 Action space: 

39 - Select node index for workload placement 

40 - Select GPU sharing ratio (0.25, 0.5, 0.75, 1.0) 

41 

42 Reward: 

43 - Positive: High GPU utilization (target 70-85%) 

44 - Positive: Low cost per workload 

45 - Positive: Meeting topology requirements 

46 - Negative: SLA violations, GPU memory OOM 

47 - Negative: Idle GPU resources 

48 """ 

49 

50 def __init__(self, num_nodes: int = 8, max_gpus_per_node: int = 8): 

51 self.num_nodes = num_nodes 

52 self.max_gpus_per_node = max_gpus_per_node 

53 

54 # State: [node_gpu_util, node_mem_util, node_cpu_util, node_gpu_free] * num_nodes 

55 # + [workload_gpu_count, workload_priority, workload_type] 

56 state_dim = num_nodes * 4 + 3 

57 self.observation_space_shape = (state_dim,) 

58 

59 # Action: node_index 

60 self.action_space_n = num_nodes 

61 

62 self.reset() 

63 

64 def reset(self) -> np.ndarray: 

65 """Reset environment to initial state.""" 

66 self.node_states = np.zeros((self.num_nodes, 4)) 

67 for i in range(self.num_nodes): 

68 self.node_states[i] = [ 

69 np.random.uniform(10, 60), # GPU utilization 

70 np.random.uniform(10, 50), # Memory utilization 

71 np.random.uniform(5, 40), # CPU utilization 

72 np.random.randint(2, self.max_gpus_per_node + 1), # Free GPUs 

73 ] 

74 

75 # Generate random workload 

76 self.current_workload = np.array( 

77 [ 

78 np.random.randint(1, 5), # GPU count needed 

79 np.random.randint(0, 100), # Priority 

80 np.random.choice([0, 1, 2]), # Type: 0=training, 1=inference, 2=fine-tuning 

81 ] 

82 ) 

83 

84 return self._get_obs() 

85 

86 def step(self, action: int) -> Tuple[np.ndarray, float, bool, Dict]: 

87 """Execute scheduling action and return reward.""" 

88 node_idx = action 

89 node = self.node_states[node_idx] 

90 gpus_needed = int(self.current_workload[0]) 

91 gpus_free = int(node[3]) 

92 

93 reward = 0.0 

94 done = False 

95 info = {} 

96 

97 if gpus_needed > gpus_free: 

98 # Cannot place - penalty 

99 reward = -10.0 

100 info["reason"] = "insufficient_gpus" 

101 else: 

102 # Place workload 

103 new_util = min(100, node[0] + gpus_needed * 12.5) 

104 

105 # Utilization reward (sweet spot: 65-85%) 

106 if 65 <= new_util <= 85: 

107 reward += 5.0 

108 elif 40 <= new_util <= 95: 

109 reward += 2.0 

110 else: 

111 reward -= 2.0 

112 

113 # Binpacking reward: prefer filling nodes 

114 if new_util > node[0]: 

115 reward += (new_util - node[0]) * 0.1 

116 

117 # Cost efficiency 

118 if self.current_workload[2] == 1: # inference 

119 reward += 1.0 # bonus for efficient inference placement 

120 

121 # Priority bonus 

122 priority = self.current_workload[1] 

123 if priority > 80: 

124 reward += 2.0 # high priority workload placed quickly 

125 

126 # Update node state 

127 self.node_states[node_idx][0] = new_util 

128 self.node_states[node_idx][3] = max(0, gpus_free - gpus_needed) 

129 

130 info["reason"] = "placed" 

131 info["node"] = node_idx 

132 info["new_utilization"] = new_util 

133 

134 # Generate next workload 

135 self.current_workload = np.array( 

136 [ 

137 np.random.randint(1, 5), 

138 np.random.randint(0, 100), 

139 np.random.choice([0, 1, 2]), 

140 ] 

141 ) 

142 

143 return self._get_obs(), reward, done, info 

144 

145 def _get_obs(self) -> np.ndarray: 

146 """Build observation vector.""" 

147 node_flat = self.node_states.flatten() 

148 return np.concatenate([node_flat, self.current_workload]).astype(np.float32)

151class SchedulingTrainer: 

152 """Trains and manages the RL scheduling model."""

154 def __init__(self, model_path: str = "./models"): 

155 self.model_path = model_path 

156 self.env = GPUSchedulingEnv() 

157 self.model = None 

158 logger.info("scheduling_trainer_initialized", model_path=model_path)

160 def train(self, total_timesteps: int = 100000, save: bool = True) -> Dict[str, Any]: 

161 """Train the scheduling optimization model.""" 

162 logger.info("training_started", total_timesteps=total_timesteps) 

163 start_time = datetime.now() 

164 

165 # Simple training loop (Q-table approach for MVP) 

166 num_episodes = total_timesteps // 100 

167 total_rewards = [] 

168 

169 q_table = np.zeros((100, self.env.action_space_n)) # simplified state discretization 

170 

171 alpha = 0.1 # learning rate 

172 gamma = 0.95 # discount factor 

173 epsilon = 1.0 # exploration rate 

174 epsilon_decay = 0.995 

175 min_epsilon = 0.01 

176 

177 for episode in range(num_episodes): 

178 obs = self.env.reset() 

179 episode_reward = 0 

180 

181 for _step in range(100): 

182 state_idx = int(np.sum(obs[:4]) % 100) # simplified state hashing 

183 

184 # Epsilon-greedy action selection 

185 if np.random.random() < epsilon: 

186 action = np.random.randint(self.env.action_space_n) 

187 else: 

188 action = np.argmax(q_table[state_idx]) 

189 

190 next_obs, reward, done, _info = self.env.step(action) 

191 next_state_idx = int(np.sum(next_obs[:4]) % 100) 

192 

193 # Q-learning update 

194 best_next = np.max(q_table[next_state_idx]) 

195 q_table[state_idx, action] += alpha * (reward + gamma * best_next - q_table[state_idx, action]) 

196 

197 episode_reward += reward 

198 obs = next_obs 

199 

200 if done: 

201 break 

202 

203 total_rewards.append(episode_reward) 

204 epsilon = max(min_epsilon, epsilon * epsilon_decay) 

205 

206 if (episode + 1) % 100 == 0: 

207 avg_reward = np.mean(total_rewards[-100:]) 

208 logger.info( 

209 "training_progress", 

210 episode=episode + 1, 

211 avg_reward=f"{avg_reward:.2f}", 

212 epsilon=f"{epsilon:.4f}", 

213 ) 

214 

215 training_time = (datetime.now() - start_time).total_seconds() 

216 

217 if save: 

218 os.makedirs(self.model_path, exist_ok=True) 

219 model_file = os.path.join(self.model_path, "scheduling_q_table.npy") 

220 np.save(model_file, q_table) 

221 logger.info("model_saved", path=model_file) 

222 

223 result = { 

224 "model": "scheduling-optimizer-v1", 

225 "algorithm": "q-learning", 

226 "episodes": num_episodes, 

227 "final_avg_reward": float(np.mean(total_rewards[-100:])), 

228 "max_reward": float(np.max(total_rewards)), 

229 "training_time_seconds": training_time, 

230 "timestamp": datetime.now().isoformat(), 

231 } 

232 

233 logger.info("training_completed", **result) 

234 return result
