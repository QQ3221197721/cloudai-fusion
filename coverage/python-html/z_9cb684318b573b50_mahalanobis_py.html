1""" 

2CloudAI Fusion - Multivariate Anomaly Detection (Mahalanobis distance). 

3 

4A real, dependency-light multivariate detector. It fits a Gaussian model 

5(mean + shrinkage-regularized covariance) to normal behavior and scores new 

6points by their squared Mahalanobis distance, which follows a chi-square 

7distribution under the fitted model. 

8 

9Unlike the per-metric (univariate) checks in detector.py, this captures JOINT 

10anomalies: correlated feature combinations that are individually in-range but 

11jointly improbable (for example a GPU at 100% utilization but only 40C, or a 

12login at a normal hour from an abnormal location+device combination). Those are 

13exactly the cases a per-metric Z-score cannot see. 

14 

15It uses numpy + scipy only (no torch/sklearn at runtime), so it is fully 

16reproducible and CI-verifiable. See benchmark.py for a labeled evaluation 

17(precision / recall / F1 / ROC-AUC) and tests/test_mahalanobis.py for the 

18CI-gated quality assertions. 

19""" 

20 

21from __future__ import annotations 

22 

23import numpy as np 

24from scipy.stats import chi2 

25 

26 

27class MahalanobisDetector: 

28 """Multivariate Gaussian anomaly detector using Mahalanobis distance. 

29 

30 fit() estimates the mean and a shrinkage-regularized covariance from 

31 normal-behavior samples; predict()/score() flag or rank new points by their 

32 squared Mahalanobis distance against a chi-square threshold. 

33 """ 

34 

35 def __init__(self, shrinkage: float = 0.1, confidence: float = 0.975): 

36 if not 0.0 <= shrinkage <= 1.0: 

37 raise ValueError("shrinkage must be in [0, 1]") 

38 if not 0.0 < confidence < 1.0: 

39 raise ValueError("confidence must be in (0, 1)") 

40 self.shrinkage = shrinkage 

41 self.confidence = confidence 

42 self.mean_: np.ndarray | None = None 

43 self.precision_: np.ndarray | None = None 

44 self.threshold_: float = 0.0 

45 self.n_features_: int = 0 

46 

47 def fit(self, x: np.ndarray) -> MahalanobisDetector: 

48 """Fit the model from normal-behavior samples x of shape (n_samples, n_features).""" 

49 data = np.asarray(x, dtype=np.float64) 

50 if data.ndim != 2: 

51 raise ValueError("x must be a 2D array of shape (n_samples, n_features)") 

52 n, k = data.shape 

53 if n < 2: 

54 raise ValueError("need at least 2 samples to estimate a covariance") 

55 self.n_features_ = k 

56 self.mean_ = data.mean(axis=0) 

57 cov = np.atleast_2d(np.cov(data, rowvar=False)) 

58 # Ledoit-Wolf-style shrinkage toward a scaled identity keeps the covariance 

59 # well-conditioned (invertible) even with few samples or collinear features. 

60 target = np.trace(cov) / k 

61 shrunk = (1.0 - self.shrinkage) * cov + self.shrinkage * target * np.eye(k) 

62 self.precision_ = np.linalg.pinv(shrunk) 

63 # Under the Gaussian model, squared Mahalanobis distance ~ chi-square(k), 

64 # so a principled threshold is the chi-square quantile at the confidence level. 

65 self.threshold_ = float(chi2.ppf(self.confidence, df=k)) 

66 return self 

67 

68 def distance(self, x: np.ndarray) -> np.ndarray: 

69 """Return the squared Mahalanobis distance of each row of x.""" 

70 if self.mean_ is None or self.precision_ is None: 

71 raise RuntimeError("detector is not fitted; call fit() first") 

72 data = np.asarray(x, dtype=np.float64) 

73 if data.ndim == 1: 

74 data = data.reshape(1, -1) 

75 centered = data - self.mean_ 

76 # d^2 = sum((centered @ precision) * centered, axis=1), vectorized. 

77 return np.einsum("ij,jk,ik->i", centered, self.precision_, centered) 

78 

79 def predict(self, x: np.ndarray) -> np.ndarray: 

80 """Return a boolean array: True where a point is an anomaly (d^2 > threshold).""" 

81 return self.distance(x) > self.threshold_ 

82 

83 def score(self, x: np.ndarray) -> np.ndarray: 

84 """Return a normalized [0, 1) anomaly score per row (0.5 at the threshold).""" 

85 d = self.distance(x) 

86 return d / (d + self.threshold_)
